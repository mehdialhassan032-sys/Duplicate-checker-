    <script>
        let globalMatrix = [];
        let originalFileName = "cleaned_infinite_sheet.xlsx";
        let isFromFile = false;
        let pendingCleanMatrix = null;

        function switchTab(tab) {
            const pasteSec = document.getElementById('section-paste');
            const sheetSec = document.getElementById('section-sheet');
            const btnPaste = document.getElementById('btn-paste');
            const btnSheet = document.getElementById('btn-sheet');

            if (tab === 'paste') {
                pasteSec.classList.remove('hidden');
                sheetSec.classList.add('hidden');
                btnPaste.className = "px-6 py-2.5 rounded-xl font-semibold bg-cyan-600 text-white transition shadow-lg shadow-cyan-600/30";
                btnSheet.className = "px-6 py-2.5 rounded-xl font-semibold bg-slate-800 text-slate-400 hover:bg-slate-700 transition";
            } else {
                pasteSec.classList.add('hidden');
                sheetSec.classList.remove('hidden');
                btnSheet.className = "px-6 py-2.5 rounded-xl font-semibold bg-cyan-600 text-white transition shadow-lg shadow-cyan-600/30";
                btnPaste.className = "px-6 py-2.5 rounded-xl font-semibold bg-slate-800 text-slate-400 hover:bg-slate-700 transition";
            }
        }

        async function triggerAutoPaste() {
            try {
                let text = await navigator.clipboard.readText();
                document.getElementById('smartPasteInput').value = text;
                processInfiniteColumnsPaste();
            } catch (err) {
                alert('Kripya textarea par click karke manually Ctrl + V dabayein.');
            }
        }

        // Helper to convert column index (0, 1, 2...) to Excel Column Letters (A, B, C... Z, AA, AB...)
        function getExcelColumnName(colIndex) {
            let columnName = '';
            let num = colIndex + 1;
            while (num > 0) {
                let remainder = (num - 1) % 26;
                columnName = String.fromCharCode(65 + remainder) + columnName;
                num = Math.floor((num - 1) / 26);
            }
            return columnName;
        }

        // Process infinite columns from pasted Google Sheet text
        function processInfiniteColumnsPaste() {
            const rawText = document.getElementById('smartPasteInput').value;
            if (!rawText.trim()) {
                alert('Kripya Google Sheets se data copy karke paste karein!');
                return;
            }

            isFromFile = true;
            originalFileName = "cleaned_sheet_data.xlsx";

            let lines = rawText.split('\n');
            let tempRows = [];
            let maxCols = 0;

            lines.forEach(line => {
                if(line.trim() !== "") {
                    let cols = line.split('\t').map(c => c.trim());
                    if(cols.length > maxCols) maxCols = cols.length;
                    tempRows.push(cols);
                }
            });

            if(tempRows.length === 0) return;

            // Generate dynamic Excel headers (Row, A, B, C... up to maxCols)
            let header = ["Row"];
            for(let i = 0; i < maxCols; i++) {
                header.push(`Col ${getExcelColumnName(i)}`);
            }

            globalMatrix = [header];
            tempRows.forEach((row, idx) => {
                let paddedRow = [idx + 1];
                for(let i = 0; i < maxCols; i++) {
                    paddedRow.push(row[i] !== undefined ? row[i] : "");
                }
                globalMatrix.push(paddedRow);
            });

            renderWorkspace(`Google Sheets Data Loaded: ${globalMatrix.length - 1} rows with ${maxCols} columns detected.`);
        }

        // Handle uploaded file with infinite columns support
        function handleInfiniteFileUpload(e) {
            const file = e.target.files[0];
            if (!file) return;
            originalFileName = "clean_" + file.name;
            isFromFile = true;

            const reader = new FileReader();
            reader.onload = function(e) {
                const data = new Uint8Array(e.target.result);
                const workbook = XLSX.read(data, {type: 'array'});
                const sheetName = workbook.SheetNames[0];
                const worksheet = workbook.Sheets[sheetName];
                
                let rawData = XLSX.utils.sheet_to_json(worksheet, {header: 1});
                if(rawData.length === 0) return;

                let maxCols = 0;
                rawData.forEach(row => {
                    if(row.length > maxCols) maxCols = row.length;
                });

                let header = ["Row"];
                for(let i = 0; i < maxCols; i++) {
                    header.push(`Col ${getExcelColumnName(i)}`);
                }

                globalMatrix = [header];
                for(let i = 0; i < rawData.length; i++) {
                    let rowData = rawData[i];
                    let paddedRow = [i + 1];
                    for(let j = 0; j < maxCols; j++) {
                        paddedRow.push(rowData[j] !== undefined ? rowData[j] : "");
                    }
                    globalMatrix.push(paddedRow);
                }

                renderWorkspace(`File Loaded: ${file.name} | Total Rows: ${globalMatrix.length - 1} | Columns: A to ${getExcelColumnName(maxCols - 1)}`);
            };
            reader.readAsArrayBuffer(file);
        }

        function renderWorkspace(statusMsg) {
            document.getElementById('workspace').classList.remove('hidden');
            document.getElementById('workspace-info').innerText = statusMsg;
            
            const table = document.getElementById('workspaceTable');
            table.innerHTML = '';

            globalMatrix.forEach((row, rIdx) => {
                let tr = document.createElement('tr');
                tr.className = rIdx === 0 ? "bg-slate-800 text-cyan-300 font-semibold sticky top-0 uppercase text-xs tracking-wider" : "border-b border-slate-800 hover:bg-slate-800/40";
                
                row.forEach((cell, cIdx) => {
                    let tag = rIdx === 0 ? 'th' : 'td';
                    let td = document.createElement(tag);
                    td.className = "px-4 py-3";
                    td.innerText = cell !== undefined && cell !== null ? cell : '';
                    if(cIdx === 0 && rIdx > 0) td.className += " text-slate-500 font-mono text-xs";
                    tr.appendChild(td);
                });
                table.appendChild(tr);
            });

            let outputPref = document.querySelector('input[name="outputType"]:checked').value;
            if(isFromFile && outputPref === 'file') {
                document.getElementById('downloadBtn').classList.remove('hidden');
            } else {
                document.getElementById('downloadBtn').classList.add('hidden');
            }
        }

        function executeInfiniteCleaning() {
            if(globalMatrix.length <= 1) return;

            let header = globalMatrix[0];
            let dataRows = globalMatrix.slice(1);

            let freq = {};
            dataRows.forEach(row => {
                for(let i = 1; i < row.length; i++) {
                    let val = String(row[i]).trim();
                    if(val.length > 3) {
                        freq[val] = (freq[val] || 0) + 1;
                    }
                }
            });

            let suspiciousVal = Object.keys(freq).find(k => freq[k] > 5);

            let uniqueMap = new Map();
            let dupCount = 0;

            dataRows.forEach(row => {
                let key = JSON.stringify(row.slice(1));
                if (!uniqueMap.has(key)) {
                    uniqueMap.set(key, row);
                } else {
                    dupCount++;
                }
            });

            let cleanedRows = Array.from(uniqueMap.values());
            cleanedRows.forEach((r, idx) => {
                r[0] = idx + 1;
            });

            pendingCleanMatrix = [header, ...cleanedRows];

            if(suspiciousVal && dupCount > 0) {
                document.getElementById('aiModalText').innerHTML = `Kuch values (jaise <span class="text-cyan-300 font-mono">"${suspiciousVal}"</span>) baar-baar same aa rahi hain. Kya yeh aapne jaan-bujh kar rakhi hain?`;
                document.getElementById('aiModal').classList.remove('hidden');
            } else {
                finalizeClean();
            }
        }

        function handleAiDecision(keep) {
            document.getElementById('aiModal').classList.add('hidden');
            if(!keep) {
                globalMatrix = pendingCleanMatrix;
                finalizeClean();
            } else {
                document.getElementById('aiStatusText').innerHTML = `<span class="font-bold text-emerald-400">Verified:</span> Identical values kept as requested.`;
            }
        }

        function finalizeClean() {
            globalMatrix = pendingCleanMatrix;
            renderWorkspace(`Cleaning Successful! Clean rows remaining: ${globalMatrix.length - 1}`);
        }

        function downloadInfiniteFile() {
            let outputPref = document.querySelector('input[name="outputType"]:checked').value;
            if(!isFromFile || outputPref !== 'file') return;

            // Remove the Row tracking index column (index 0) before export
            let exportMatrix = globalMatrix.map(row => row.slice(1));
            let ws = XLSX.utils.aoa_to_sheet(exportMatrix);
            let wb = XLSX.utils.book_new();
            XLSX.utils.book_append_sheet(wb, ws, "CleanedData");
            XLSX.writeFile(wb, originalFileName);
        }
    </script>
